class Solution {
    public int maxProfit(int k, int[] prices) {
        int  n = prices.length;

    int[][][] dp = new int[n+1][2][k+1];

    for (int i = 0; i <= n; i++) {
            for (int buy = 0; buy < 2; buy++) {
                dp[i][buy][0] = 0;
            }
        }

        for (int buy = 0; buy < 2; buy++) {
            for (int i = 0; i <= k; i++) {
                dp[n][buy][i] = 0;
            }
        }


        for(int i = n-1 ; i >= 0 ; i--){
            for (int buy = 0; buy < 2; buy++){
                for(int j = 1 ; j < k+1 ; j++){
                    if(buy==1){
                        dp[i][buy][j] = Math.max(dp[i+1][0][j]-prices[i],dp[i+1][1][j]);
                    }else{
                         dp[i][buy][j] = Math.max(dp[i+1][1][j-1]+prices[i],dp[i+1][0][j]);
                    }

                }
            }
        }

        return dp[0][1][k];
    }
}
