class Solution {
    public boolean isMatch(String s, String p) {
        int n = s.length();
        int m = p.length();

        boolean[][] dp = new boolean[n + 1][m + 1];

        dp[0][0] = true;
        for (int j = 1; j <= n; j++) {
            dp[j][0] = false;
        }
        for (int i = 1; i <= m; i++) {
            boolean flag = true;
            for (int ii = 1; ii <= i; ii++) {
                if (p.charAt(ii - 1) != '*') {
                    flag = false;
                    break;
                }
            }

            dp[0][i] = flag;

        }

        for (int i = 1; i <= n; i++) {
            for (int j = 1; j <= m; j++) {
                if (s.charAt(i - 1) == p.charAt(j - 1) || p.charAt(j - 1) == '?') {
                    dp[i][j] = dp[i - 1][j - 1];
                } else if (p.charAt(j - 1) == '*') {
                    dp[i][j] = dp[i - 1][j] || dp[i][j - 1];
                } else {
                    dp[i][j] = false;
                }

            }
        }

        return dp[n][m];

    }
}


import java.util.*;

// Class containing the logic for wildcard pattern matching
class Solution {

    // Helper function to check if all characters in S1[0...i-1] are '*'
    public boolean isAllStars(String S1, int i) {
        for (int j = 1; j <= i; j++) {
            if (S1.charAt(j - 1) != '*') return false;
        }
        return true;
    }

    // Function to perform wildcard matching
    public boolean wildcardMatching(String S1, String S2) {
        int n = S1.length();
        int m = S2.length();

        // Previous row results
        boolean[] prev = new boolean[m + 1];
        // Current row results
        boolean[] cur = new boolean[m + 1];

        // Empty pattern matches empty string
        prev[0] = true;

        // Iterate over each character of pattern
        for (int i = 1; i <= n; i++) {
            // Check if first column should be true
            cur[0] = isAllStars(S1, i);

            // Iterate over each character of text
            for (int j = 1; j <= m; j++) {
                char pChar = S1.charAt(i - 1);
                char tChar = S2.charAt(j - 1);

                if (pChar == tChar || pChar == '?') {
                    cur[j] = prev[j - 1];
                } else if (pChar == '*') {
                    cur[j] = prev[j] || cur[j - 1];
                } else {
                    cur[j] = false;
                }
            }

            // Move current row to previous
            prev = Arrays.copyOf(cur, m + 1);
        }

        return prev[m];
    }
}

// Driver code
public class Main {
    public static void main(String[] args) {
        Solution sol = new Solution();
        String S1 = "ab*cd";
        String S2 = "abdefcd";

        if (sol.wildcardMatching(S1, S2))
            System.out.println("String S1 and S2 do match");
        else
            System.out.println("String S1 and S2 do not match");
    }
}
