class Solution {

    static Boolean isSubsetSum(int arr[], int sum) {
        // code here
        boolean[] prev = new boolean[sum+1];
        prev[0] = true;
        // curr[0] = true;
        int n = arr.length;
        
        if(sum>=arr[0])
        prev[arr[0]]=true;
        
        for(int i = 1 ; i < n ;i++){
            boolean[] curr = new boolean[sum+1];
            curr[0] = true;
            for(int tar = 1 ;tar <= sum ;tar++){
                
                boolean nottake = prev[tar];
                boolean take = false;
                if(tar>=arr[i]){
                    take = prev[tar-arr[i]];
                }
                
                curr[tar] = take||nottake;
            }
            prev = curr;
            
            
        }
        return prev[sum];
                
    }
    public boolean canPartition(int[] nums) {
        


        int totSum =  0;
        for(int n : nums){
            totSum += n;
        }

        if(totSum%2==1) return false;
        int k = totSum/2;

        return isSubsetSum(nums,k);


    }
}
