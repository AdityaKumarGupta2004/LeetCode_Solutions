class Solution {
	static int helper(int arr[], int ind, int prev, int dp[][]) {
		if (ind == arr.length)
			return 0;
		if (dp[ind][prev + 1] != -1)
			return dp[ind][prev + 1];
		
		int len = helper(arr, ind + 1, prev, dp);
		
		if (prev == -1 || arr[ind] > arr[prev]) {
			len = Math.max(1 + helper(arr, ind + 1, ind, dp), len);
			
		}
		dp[ind][prev + 1] = len;
		
		return len;
	}
	public int lis(int arr[]) {
		// for(int[] a: dp){
		//     Arrays.fill(a,-1);
		// }
		
		// return helper(arr,0,-1,dp);
		
		int n = arr.length;
		
		int dp[][] = new int[n + 1][n + 1];
		
		// Base case already initialized
		
		for (int i = n - 1 ; i >= 0 ; i--) {
			for (int ind = i - 1 ; ind >= -1 ; ind--) {
				int len = dp[i + 1][ind+1];
				if ( ind == -1 || arr[i] > arr[ind]) {
					len = Math.max(1 + dp[i + 1][i+1], len);
				}
				dp[i][ind+1] = len ;
			}
		}
		
		return dp[0][0];
		
	}
}
