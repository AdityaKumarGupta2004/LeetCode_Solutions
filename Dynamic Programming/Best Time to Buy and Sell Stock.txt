class Solution {
    public int maxProfit(int[] prices) {
        int n = prices.length;
        int mini = prices[0] ;
        int pro = 0;
        for(int i  = 1  ;  i < n ; i++){
            int cost = prices[i] - mini;
            pro = Math.max(cost,pro);

            mini =  Math.min(mini,prices[i]);
        }

        return pro;


    }
}
